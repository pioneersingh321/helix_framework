const useCartStore = Helix.defineStore('cart', {
  state: () => ({ items: [], discountRate: 0.20 }),
  getters: {
    subtotal(state) { return state.items.reduce((acc, i) => acc + (i.price * i.qty), 0); },
    grandTotal(state) { return (this.subtotal - this.discountAmount) + this.taxAmount; }
  },
  history: true
});
Tier 3

3. Composable / Setup Store (User Auth & Permissions)

Helix.defineStore(id, () => { ... })

Write Pinia-style composable stores using standard Helix.ref() and Helix.computed(). All returned refs, computeds, and action functions are automatically bound to the store proxy.

Current User Session
Computed Capabilities (store.auth.canDeploy):
Deploy Production Manage Users Edit Schema
const useAuthStore = Helix.defineStore('auth', () => {
  const user = Helix.ref({ name: 'Sarah', role: 'admin' });
  const isAuthenticated = Helix.computed(() => user.value !== null);
  const canDeploy = Helix.computed(() => user.value?.role === 'admin');
  const canManageUsers = Helix.computed(() => ['admin', 'architect'].includes(user.value?.role));

  function login(name, role, email) {
    user.value = { name, role, email };
  }

  return { user, isAuthenticated, canDeploy, canManageUsers, login };
});
