const users = [
  { id: 1, name: 'Alice', dept: 'Eng', salary: 95000 },
  { id: 2, name: 'Bob', dept: 'Design', salary: 80000 },
  { id: 3, name: 'Charlie', dept: 'Eng', salary: 110000 },
  { id: 4, name: 'Dana', dept: 'Product', salary: 90000 }
];

$h.groupBy(users, 'dept');           // { Eng: [Alice, Charlie], Design: [Bob], ... }
$h.keyBy(users, 'id');               // { 1: Alice, 2: Bob, 3: Charlie, 4: Dana }
$h.sortBy(users, 'salary', 'desc');  // [Charlie, Alice, Dana, Bob]
$h.chunk(users, 2);                  // [[Alice, Bob], [Charlie, Dana]]
$h.pluck(users, 'name');             // ['Alice', 'Bob', 'Charlie', 'Dana']
$h.findBy(users, 'dept', 'Design');  // Bob object
$h.removeBy(users, 'id', 2);         // Array without Bob
$h.partition(users, u => u.salary >= 90000); // [[Alice, Charlie, Dana], [Bob]]
$h.difference([1, 2, 3], [2, 4]);    // [1, 3]
$h.intersection([1, 2, 3], [2, 3]);  // [2, 3]
$h.unique([1, 1, 2, 3, 3]);          // [1, 2, 3]
$h.flatten([[1, [2]], [3]], 2);      // [1, 2, 3]
