// Password strength check
const state = reactive({
  password: '',
  get passwordStrength() {
    const len = this.password.length;
    if (len === 0) return 'None';
    if (len < 6) return 'Weak';
    if (len < 10) return 'Medium';
    return 'Strong';
  },
  get passwordClass() {
    const str = this.passwordStrength;
    if (str === 'Weak') return 'text-danger';
    if (str === 'Medium') return 'text-warning';
    return 'text-success';
  }
});

Custom Directive Registration

// Debugger Directive Setup
app.directive('debug', {
  mounted(el, binding) {
    const { value, ctx, trackCleanup } = binding;
    const handler = () => {
      console.log("Context Data:", ctx);
    };
    el.addEventListener('click', handler);
    trackCleanup(() => 
      el.removeEventListener('click', handler)
    );
  }
});
