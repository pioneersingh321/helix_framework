// Password strength check
const state = reactive({
  password: '',
  get passwordStrength() {
    const len = this.password.length;
    if (len === 0) return 'None';
    if (len < 6) return 'Weak';
    if (len < 10) return 'Medium';
    return 'Strong';
  },
  get passwordClass() {
    const str = this.passwordStrength;
    if (str === 'Weak') return 'text-danger';
    if (str === 'Medium') return 'text-warning';
    return 'text-success';
  }
});

    const { value, ctx, trackCleanup } = binding;

          Section 10
          🎯 DOM Events & Vue-Style Modifiers
        

        

          Helix Core provides Vue-style DOM event modifiers with a unified event handling pipeline. Modifiers filter events by key, mouse button, system keys, exact match, self-target, and configure native listener lifecycle options.
        


        

          


            
            

              
1. Keyboard Modifiers

              

                Filters key events by specific keys (e.g. .enter, .tab, .delete, .esc, .space, .up/.down/.left/.right) or dynamic kebab-case keys (.page-down).
              

              
              

                
              

            


            
            

              
2. Mouse Modifiers

              

                Filter mouse clicks by button: .left (primary), .middle (auxiliary / scroll wheel), .right (secondary with .prevent).
              

              

                
                
                
              

              

                
              

            


            
            

              
3. System Modifiers & .exact

              

                Enforce modifier keys: .ctrl, .alt, .shift, .meta, and .exact. Note that .prevent & .stop only trigger if the filter succeeds!
              

              

                
                
                
              

              

                
              

            


            
            

              
4. Event Flow & Native Options (.self, .once, .stop, .prevent)

              

                .self prevents triggers from child clicks; .once registers with native listener option to execute only once.
              

              

                Outer Box (has @click.self)
                
                
              

              
              

                
              

            


          


          

            
Event Directives Cheat Sheet

            
<!-- 1. Keyboard Modifiers -->
<input @keydown.enter="submit" />
<input @keydown.tab="onTab" />
<input @keydown.delete="onDeleteOrBackspace" />
<input @keydown.esc="closeModal" />
<input @keydown.space="onSpace" />
<input @keydown.up="navigateUp" />
<input @keydown.down="navigateDown" />
<!-- Dynamic Kebab-Case Keys -->
<input @keydown.page-down="nextPage" />

<!-- 2. Mouse Modifiers -->
<button @click.left="primaryAction">Left</button>
<button @click.middle="auxAction">Middle</button>
<button @click.right.prevent="contextMenu">Right</button>

<!-- 3. System Modifier Keys & .exact -->
<!-- Only prevents default if Ctrl is actually held -->
<button @click.ctrl.prevent="save">Save</button>
<!-- Only triggers if ONLY Ctrl is pressed (no Shift/Alt) -->
<button @click.ctrl.exact="doSomething">Ctrl Only</button>
<!-- Only triggers when NO system modifier is pressed -->
<button @click.exact="plainClick">No Keys</button>

<!-- 4. Event Flow & Native Options -->
<!-- Only triggers if event.target === el -->
<div @click.self="onBackdropClick">...</div>
<!-- Native options passed to addEventListener -->
<button @click.once="initOnce">Init</button>
<div @scroll.passive="onScroll">...</div>
<div @click.capture="onCapture">...</div>

// 1. Register Reusable Modules
Helix.module('cart', {
  price: 1000,
  total() {
    const tax = this.$app.module('tax');
    return this.price + tax.calculate(this.price);
  }
});

Helix.module('tax', {
  rate: 18,
  calculate(amount) {
    return (amount * this.rate) / 100;
  }
});

// 2. Direct In-Template Usage
<div>
  <p>Tax rate: [[ tax.rate ]]%</p>
  <p>Tax amount: $[[ tax.calculate(1000) ]]</p>
  <p>Total: $[[ cart.total() ]]</p>
</div>

// 3. Mutate Globally Anywhere
Helix.module('tax').rate = 12;

// 4. Restore Baseline Snapshot
Helix.module('cart').$reset();
